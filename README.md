class Developer {

    std::vector<std::string> languages;
    std::vector<std::string> interests;
    
public:
    Developer(std::string name, std::string location) :

        languages = {"C++", "Java", "Python", "JavaScript", "C"};

    void greet() {

        std::cout << "Currently studying CS at Western University.";

};
